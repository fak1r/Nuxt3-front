-- Catalog presentation data. On a clean database these columns are created by
-- the SQLAlchemy models; for an existing database use backend/migrations/20261120_catalog_presentation.sql.
UPDATE producers
SET
  country = CASE slug
    WHEN 'classen' THEN 'Германия'
    WHEN 'pergo' THEN 'Швеция'
    ELSE country
  END,
  logo_url = CASE slug
    WHEN 'classen' THEN 'catalog/producers/classen/logo.svg'
    WHEN 'pergo' THEN 'catalog/producers/pergo/logo.webp'
    ELSE logo_url
  END
WHERE slug IN ('classen', 'pergo');
